# В scripts/hermes_hooks.py добавить
from dory import DoryMemory
mem = DoryMemory(db_path="./hermes_knowledge.db")

def on_task_complete(issue_id, fix_result):
    if fix_result.get("verified"):
        mem.observe(
            f"Fix for {issue_id}: {fix_result['patch']}",
            node_type="SOLUTION",
            importance=4  # critical
        )
        mem.observe(
            f"Verification: {fix_result['evidence']}",
            node_type="EVIDENCE"
        )
