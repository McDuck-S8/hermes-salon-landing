8. Скрипт парсинга конкурентов (MVP)

#!/usr/bin/env python3
"""Парсинг Telegram каналов-конкурентов для контент-матрицы"""
import requests, json, time, re

CHANNELS = ["cryptosignals", "arbitrage_channel", "cpa_traffic"]
# Анализируем: формат постов, частоту, вовлечение, темы

for ch in CHANNELS:
    url = f"https://t.me/s/{ch}"
    resp = requests.get(url, timeout=10)
    # Парсим HTML — извлекаем текст, медиа, даты
    posts = re.findall(r'<div class="tgme_widget_message_text">(.*?)</div>', resp.text)
    print(f"{ch}: {len(posts)} постов")
