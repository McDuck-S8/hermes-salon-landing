Python Bayesian A/B Test with PyMC - Beta-Binomial Implementation

```python
import pymc as pm
import numpy as np
import arviz as az

control_conversions, control_visitors = 120, 5000
treatment_conversions, treatment_visitors = 150, 5000

with pm.Model() as model:
    p_control = pm.Beta("p_control", alpha=1, beta=1)
    p_treatment = pm.Beta("p_treatment", alpha=1, beta=1)
    obs_control = pm.Binomial("obs_control", n=control_visitors, p=p_control, observed=control_conversions)
    obs_treatment = pm.Binomial("obs_treatment", n=treatment_visitors, p=p_treatment, observed=treatment_conversions)
    diff = pm.Deterministic("diff", p_treatment - p_control)
    rel_lift = pm.Deterministic("rel_lift", (p_treatment - p_control) / p_control)
    trace = pm.sample(2000, tune=1000, target_accept=0.95, return_inferencedata=True)

prob_superior = (trace.posterior["diff"] > 0).mean().item()
print(f"P(treatment > control) = {prob_superior:.3f}")
loss = (trace.posterior["diff"].clip(max=0)).mean().item()
print(f"Expected loss = {loss:.5f}")
hdi = az.hdi(trace.posterior["diff"], hdi_prob=0.95)
print(f"95% HDI: {hdi.values}")
```

Beta(1,1) = uniform. Skeptical prior: Beta(10, 990) for ~1% baseline. Check R-hat < 1.01, ESS > 400.
