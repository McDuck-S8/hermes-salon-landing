Sequential Testing - Always-Valid Inference

mSPRT - Always-Valid p-values:
```python
import numpy as np
from scipy.stats import chi2

def msprt_pvalue(control_conv, control_vis, treatment_conv, treatment_vis, prior_var=0.25):
    p1 = control_conv / control_vis
    p2 = treatment_conv / treatment_vis
    n1, n2 = control_vis, treatment_vis
    var_diff = p1*(1-p1)/n1 + p2*(1-p2)/n2
    z = (p2 - p1) / np.sqrt(var_diff)
    lambda_ = 1 / (1 + var_diff / prior_var)
    msprt_stat = lambda_ * z**2 / (1 + lambda_)
    pval = 1 - chi2.cdf(msprt_stat, df=1)
    return pval
```

def obf_boundary(alpha=0.05, info_frac=0.5):
    from scipy.stats import norm
    z = norm.ppf(1 - alpha/2)
    return z / np.sqrt(info_frac)
