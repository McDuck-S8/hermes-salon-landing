Bayesian Power Analysis - Simulation-Based Sample Size

```python
import numpy as np
import pymc as pm

def simulate_power(true_lift=0.02, baseline=0.02, n_per_group=5000, n_sim=500):
    wins = 0
    for _ in range(n_sim):
        control_conv = np.random.binomial(n_per_group, baseline)
        treatment_conv = np.random.binomial(n_per_group, baseline * (1 + true_lift))
        with pm.Model():
            p_c = pm.Beta("p_c", 1, 1)
            p_t = pm.Beta("p_t", 1, 1)
            pm.Binomial("obs_c", n=n_per_group, p=p_c, observed=control_conv)
            pm.Binomial("obs_t", n=n_per_group, p=p_t, observed=treatment_conv)
            trace = pm.sample(1000, tune=500, chains=2, progressbar=False)
        prob_superior = (trace.posterior["p_t"] > trace.posterior["p_c"]).mean()
        if prob_superior > 0.95:
            wins += 1
    return wins / n_sim

for n in [2000, 3000, 4000, 5000, 6000]:
    power = simulate_power(true_lift=0.02, baseline=0.02, n_per_group=n, n_sim=100)
    print(f"n={n}: power={power:.2f}")

def assurance(prior_lift_mean=0.02, prior_lift_sd=0.01, n=5000):
    lifts = np.random.normal(prior_lift_mean, prior_lift_sd, 100)
    powers = [simulate_power(l, 0.02, n, 50) for l in lifts if l > 0]
    return np.mean(powers)

print(f"Assurance at n=5000: {assurance():.2f}")
```

Classical power assumes fixed effect; Bayesian integrates over prior (assurance). Add 20-30% buffer for traffic imbalance, bots, seasonality.
