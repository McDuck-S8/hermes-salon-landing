Отлично. Переходим к **автоматизации реферальных программ** — это уже enterprise-уровень. Речь идёт не о том, КАК привлекать, а о том, КАК автоматически отслеживать, верифицировать, платить и детектить фрод.

## 4. Интеграционные паттерны

### 4.1 Postback / Webhook (универсальный)

**Что такое postback:** Это HTTP-запрос от вашего сайта/приложения к платформе (Tapfiliate и др.), который сообщает о конверсии.

**Схема:**
```
Пользователь кликнул по ссылке → Попал на ваш сайт → Совершил целевое действие (регистрация/оплата) → Ваш сервер отправляет POST-запрос на платформу → Платформа начисляет комиссию
```

**Пример webhook для Tapfiliate (Node.js):**

```javascript
const express = require('express');
const axios = require('axios');
const app = express();
app.use(express.json());

// Webhook для обработки конверсии (например, после оплаты в Stripe)
app.post('/webhook/referral-conversion', async (req, res) => {
    const { userId, email, amount, currency } = req.body;

    // 1. Проверяем, есть ли реферальная метка у пользователя
    const referralData = await getReferralFromDB(userId);
    
    if (!referralData) {
        return res.status(200).json({ status: 'no_referral' });
    }

    // 2. Отправляем postback в Tapfiliate
    const tapfiliatePayload = {
        conversion: {
            external_id: `order_${Date.now()}`,
            amount: amount,
            currency: currency,
            email: email,
            status: 'approved' // или 'pending' для ручной проверки
        }
    };

    try {
        await axios.post(
            `https://api.tapfiliate.com/1.6/conversions/`,
            tapfiliatePayload,
            {
                headers: {
                    'Authorization': 'Bearer YOUR_TAPFILIATE_API_KEY',
                    'Content-Type': 'application/json'
                }
            }
        );

        // 3. Логируем
        console.log(`[Referral] Конверсия зафиксирована: ${email} - $${amount}`);
        
        // 4. Отправляем уведомление партнёру (Telegram/Email)
        await sendNotification(referralData.partnerId, `Новая конверсия: $${amount}`);

        res.status(200).json({ status: 'success' });
    } catch (error) {
        console.error('Tapfiliate error:', error.response.data);
        res.status(500).json({ status: 'error', message: error.message });
    }
});

app.listen(3000, () => console.log('Webhook server running on port 3000'));
```

### 4.2 UTM-трекинг (для всех платформ)

**Стандартные UTM-метки для рефералов:**

```
https://yourwebsite.com/?utm_source={partner_name}&utm_medium=referral&utm_campaign={campaign_name}&utm_content={content_type}&ref={partner_id}
```

**Пример:**
```
https://saas.com/?utm_source=alex_blog&utm_medium=referral&utm_campaign=q4_2024&utm_content=review_post&ref=ALEX2024
```

**Как парсить UTM в Tapfiliate:**

```javascript
// В момент регистрации пользователя
function captureUTMParams() {
    const urlParams = new URLSearchParams(window.location.search);
    const utmData = {
        utm_source: urlParams.get('utm_source'),
        utm_medium: urlParams.get('utm_medium'),
        utm_campaign: urlParams.get('utm_campaign'),
        utm_content: urlParams.get('utm_content'),
        ref: urlParams.get('ref')
    };

    // Сохраняем в localStorage
    localStorage.setItem('utm_data', JSON.stringify(utmData));

    // Отправляем на сервер при регистрации
    return utmData;
}

// Отправка при регистрации
async function registerUser(email, password) {
    const utmData = JSON.parse(localStorage.getItem('utm_data') || '{}');
    
    const response = await fetch('/api/register', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            email,
            password,
            utm: utmData,
            referrer: utmData.ref || null
        })
    });

    return response.json();
}
```

### 4.3 Антифрод-система (многослойная)

**Уровни защиты:**

| Уровень | Метод | Что проверяет |
|---------|-------|---------------|
| **1. Базовый** | IP-адрес | Один IP не может создать >5 аккаунтов за 24ч |
| **2. Email** | Проверка домена | Временные email (tempmail, 10minute) — блок |
| **3. Поведенческий** | Время между действиями | Клик → регистрация → оплата < 2 минут — подозрительно |
| **4. Куки** | Проверка согласованности | Куки должны совпадать с User-Agent |
| **5. Гео** | Страна/IP | Если страна не совпадает с платежом — флаг |

**Реализация (Node.js + Redis):**

```javascript
const redis = require('redis');
const client = redis.createClient();

class FraudDetector {
    constructor() {
        this.redis = client;
    }

    // 1. Проверка IP (лимит регистраций)
    async checkIPLimit(ip) {
        const key = `ip:${ip}:registrations`;
        const count = await this.redis.incr(key);
        if (count === 1) {
            await this.redis.expire(key, 86400); // 24 часа
        }
        return count <= 5; // максимум 5 регистраций с одного IP
    }

    // 2. Проверка email на временный домен
    checkTempEmail(email) {
        const tempDomains = [
            'tempmail.com', '10minutemail.com', 'guerrillamail.com',
            'mailinator.com', 'yopmail.com', 'throwawaymail.com'
        ];
        const domain = email.split('@')[1];
        return !tempDomains.includes(domain);
    }

    // 3. Проверка скорости (от клика до конверсии)
    async checkConversionSpeed(sessionId) {
        const key = `session:${sessionId}:click_time`;
        const clickTime = await this.redis.get(key);
        
        if (!clickTime) return false; // нет клика → фрод
        
        const now = Date.now();
        const diffMinutes = (now - parseInt(clickTime)) / (1000 * 60);
        
        // Меньше 2 минут → подозрительно (только для высоких сумм)
        return diffMinutes >= 2;
    }

    // 4. Комплексная проверка
    async validateConversion(data) {
        const { ip, email, sessionId, amount } = data;

        const checks = {
            ipOk: await this.checkIPLimit(ip),
            emailOk: this.checkTempEmail(email),
            speedOk: await this.checkConversionSpeed(sessionId)
        };

        // Логируем результат
        console.log('[Fraud Check]', checks);

        // Если все проверки пройдены
        if (checks.ipOk && checks.emailOk && checks.speedOk) {
            return { approved: true };
        }

        // Если есть сомнения - отправляем на ручную проверку
        if (!checks.speedOk && amount > 100) {
            return { 
                approved: false, 
                reason: 'Слишком быстрая конверсия (возможный фрод)',
                details: checks
            };
        }

        return { 
            approved: false, 
            reason: 'Не пройдена антифрод-проверка',
            details: checks
        };
    }
}

// Использование
const fraud = new FraudDetector();
const result = await fraud.validateConversion({
    ip: '192.168.1.1',
    email: 'user@gmail.com',
    sessionId: 'abc123',
    amount: 199
});

if (!result.approved) {
    // Отправить в ручную проверку или отклонить
    console.log('Fraud detected:', result.reason);
}
```

**Вариант 1: PayPal Payouts (для Tapfiliate / FirstPromoter)**

```javascript
const axios = require('axios');

async function processPayout(partnerEmail, amount, currency = 'USD') {
    // PayPal API (используем SDK или прямой REST)
    const paypalResponse = await axios.post(
        'https://api-m.sandbox.paypal.com/v1/payments/payouts',
        {
            sender_batch_header: {
                sender_batch_id: `batch_${Date.now()}`,
                email_subject: 'Выплата по реферальной программе',
                email_message: 'Спасибо!'
            },
            items: [{
                recipient_type: 'EMAIL',
                amount: {
                    value: amount.toFixed(2),
                    currency: currency
                },
                receiver: partnerEmail,
                note: `Реферальная комиссия за ${new Date().toLocaleDateString()}`
            }]
        },
        {
            headers: {
                'Authorization': `Bearer ${process.env.PAYPAL_ACCESS_TOKEN}`,
                'Content-Type': 'application/json'
            }
        }
    );

    // Логируем выплату
    await savePayoutRecord(partnerEmail, amount, paypalResponse.data.batch_header.payout_batch_id);
    
    return paypalResponse.data;
}
```

const stripe = require('stripe')(process.env.STRIPE_SECRET_KEY);

}

## 11. Ошибки при внедрении (и как избежать)

| Ошибка | Последствие | Решение |
|--------|-------------|---------|
| Не настраивать webhook на все события | Потеря конверсий | Добавить обработку: регистрация, оплата, отмена, возврат |
| Слишком низкий порог антифрода | Блокировка реальных партнёров | Настроить гибкие правила (разные для малых/крупных сумм) |
| Выплаты вручную | Затраты времени, ошибки | Полностью автоматизировать через API PayPal/Stripe |

## 12. Бонус: скрипт для миграции партнёров (из CSV в Tapfiliate)

```python
import csv
import requests

API_KEY = 'your_tapfiliate_api_key'
URL = 'https://api.tapfiliate.com/1.6/affiliates/'

def migrate_partners(csv_file):
    with open(csv_file, 'r') as file:
        reader = csv.DictReader(file)
        for row in reader:
            payload = {
                'firstname': row['first_name'],
                'lastname': row['last_name'],
                'email': row['email'],
                'status': 'active',
                'metadata': {
                    'source': row['source'],
                    'joined_date': row['joined_date']
                }
            }
            
            response = requests.post(URL, json=payload, 
                                    headers={'Authorization': f'Bearer {API_KEY}'})
            
            if response.status_code == 201:
                print(f"✅ Партнёр {row['email']} импортирован")
            else:
                print(f"❌ Ошибка: {response.text}")

# Использование
migrate_partners('partners.csv')
```
