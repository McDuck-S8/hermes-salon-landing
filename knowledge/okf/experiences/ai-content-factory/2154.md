# Блок 1. Архитектура продакшен-системы на ComfyUI

**Цель:** Создать API-сервис, который принимает запросы, генерирует изображения и возвращает результат, масштабируясь до тысяч генераций в день.

## 1.1. Компоненты системы
```
[Клиент] → [API Gateway (FastAPI)] → [Redis Queue] → [ComfyUI Workers] → [S3 Storage] → [Клиент]
```
- **API Gateway** — принимает запросы, валидирует промпты, сохраняет в очередь.
- **Redis** — хранит задания и статусы.
- **Workers** — экземпляры ComfyUI с разными моделями.
- **S3** — финальное хранилище результатов.

## 3.2. Python-скрипт для батч-обработки (через API)

```python
import requests
import json
import base64
import time
from typing import Dict, List

class ComfyUIBatchProcessor:
    def __init__(self, url="http://localhost:8188"):
        self.url = url
        self.client_id = "batch_processor"
    
    def load_workflow_template(self, template_path: str) -> Dict:
        """Загрузка базового workflow-шаблона"""
        with open(template_path, 'r') as f:
            return json.load(f)
    
    def build_prompt(self, template: Dict, params: Dict) -> Dict:
        """Динамическая подстановка промптов в workflow"""
        # Находим ноду с текстом (обычно это CLIP Text Encode)
        for node_id, node in template.items():
            if node.get('class_type') == 'CLIPTextEncode':
                # Подставляем параметры
                text = node['inputs']['text']
                for key, value in params.items():
                    text = text.replace(f"{{{key}}}", value)
                node['inputs']['text'] = text
        return template
    
    def queue_prompt(self, prompt: Dict) -> str:
        """Отправка промпта в очередь"""
        data = {
            "prompt": prompt,
            "client_id": self.client_id
        }
        response = requests.post(f"{self.url}/prompt", json=data)
        if response.status_code != 200:
            raise Exception(f"Error queueing prompt: {response.text}")
        return response.json()['prompt_id']
    
    def process_batch(self, batch_config: Dict) -> List[str]:
        """Обработка всего батча"""
        template = self.load_workflow_template(batch_config.get('template', 'workflow_api.json'))
        prompt_ids = []
        
        for prompt_data in batch_config['prompts']:
            # Строим промпт с подстановкой
            workflow = self.build_prompt(template, prompt_data)
            # Добавляем настройки модели
            workflow = self.update_model_settings(workflow, batch_config)
            # Отправляем в очередь
            prompt_id = self.queue_prompt(workflow)
            prompt_ids.append(prompt_id)
            print(f"✅ Задание {prompt_data['id']} отправлено, ID: {prompt_id}")
            time.sleep(0.1)  # небольшая задержка
        
        return prompt_ids
    
    def update_model_settings(self, workflow: Dict, config: Dict) -> Dict:
        """Обновление настроек модели (шаги, CFG, размер)"""
        for node_id, node in workflow.items():
            if node.get('class_type') == 'KSampler':
                node['inputs']['steps'] = config.get('steps', 30)
                node['inputs']['cfg'] = config.get('cfg', 7.5)
            if node.get('class_type') == 'EmptyLatentImage':
                node['inputs']['width'] = config.get('width', 1024)
                node['inputs']['height'] = config.get('height', 1024)
        return workflow

# Использование
processor = ComfyUIBatchProcessor()
with open('batch_config.json', 'r') as f:
    config = json.load(f)

prompt_ids = processor.process_batch(config)
print(f"✅ Отправлено {len(prompt_ids)} заданий")
```

```python
style_variants = {
    "minimal": {
        "positive": "минимализм, чистые линии, нейтральные цвета",
        "negative": "шум, перегруженность"
    },
    "vintage": {
        "positive": "винтаж, ретро, зернистость, теплые тона",
        "negative": "современный, глянец"
    },
    "cinematic": {
        "positive": "кинематографичный, драматическое освещение, глубина резкости",
        "negative": "плоское освещение"
    }
}

def apply_style(workflow, style_name):
    style = style_variants[style_name]
    for node_id, node in workflow.items():
        if node.get('class_type') == 'CLIPTextEncode':
            # Определяем, положительный это промпт или отрицательный
            if 'positive' in node_id or node['inputs'].get('text', '').startswith('фотореалистичный'):
                node['inputs']['text'] += f", {style['positive']}"
            else:
                node['inputs']['text'] += f", {style['negative']}"
    return workflow
```

```python
class BatchProcessorNode:
    """Нода для обработки батча с динамическими промптами"""
    
    @classmethod
    def INPUT_TYPES(cls):
        return {
            "required": {
                "model": ("MODEL",),
                "positive_base": ("STRING", {"multiline": True}),
                "negative_base": ("STRING", {"multiline": True}),
            },
            "hidden": {
                "prompt": "PROMPT",
                "extra_pnginfo": "EXTRA_PNGINFO"
            }
        }
    
    RETURN_TYPES = ("LATENT",)
    FUNCTION = "process_batch"
    CATEGORY = "batch_processing"
    
    def process_batch(self, model, positive_base, negative_base, prompt=None, extra_pnginfo=None):
        # Парсим параметры из промпта
        batch_params = self.parse_batch_params(prompt)
        
        latents = []
        for params in batch_params:
            # Генерируем промпт для каждого элемента
            positive = positive_base.format(**params)
            negative = negative_base.format(**params)
            
            # Здесь вызов стандартного KSampler
            latent = self.generate(model, positive, negative)
            latents.append(latent)
        
        # Возвращаем батч латентов
        return (torch.cat(latents, dim=0),)
```

# Блок 6. Управление очередью и мониторинг

## 6.1. API-эндпоинты для управления очередью

```python
# queue_manager.py
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import requests
import json
from typing import Optional

app = FastAPI()
COMFYUI_URL = "http://localhost:8188"

class GenerationRequest(BaseModel):
    workflow: dict
    priority: Optional[int] = 0
    callback_url: Optional[str] = None

@app.post("/generate")
async def generate_image(request: GenerationRequest):
    """Отправка задания в ComfyUI"""
    try:
        data = {
            "prompt": request.workflow,
            "client_id": "api_gateway"
        }
        response = requests.post(f"{COMFYUI_URL}/prompt", json=data)
        
        if response.status_code == 200:
            prompt_id = response.json()['prompt_id']
            return {
                "status": "queued",
                "prompt_id": prompt_id,
                "queue_position": get_queue_position(prompt_id)
            }
        else:
            raise HTTPException(status_code=500, detail="ComfyUI error")
    
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/status/{prompt_id}")
async def get_status(prompt_id: str):
    """Проверка статуса задания"""
    response = requests.get(f"{COMFYUI_URL}/history")
    if response.status_code == 200:
        history = response.json()
        if prompt_id in history:
            return {
                "status": "completed",
                "output": history[prompt_id]['outputs']
            }
        else:
            # Проверяем, не в очереди ли еще
            queue_response = requests.get(f"{COMFYUI_URL}/queue")
            queue = queue_response.json()
            if prompt_id in queue.get('queue_running', []):
                return {"status": "running"}
            elif prompt_id in queue.get('queue_pending', []):
                return {"status": "pending"}
            else:
                return {"status": "not_found"}
    else:
        raise HTTPException(status_code=500, detail="ComfyUI error")

def get_queue_position(prompt_id: str) -> int:
    """Получение позиции в очереди"""
    response = requests.get(f"{COMFYUI_URL}/queue")
    if response.status_code == 200:
        queue = response.json()
        pending = queue.get('queue_pending', [])
        for idx, item in enumerate(pending):
            if item[0] == prompt_id:
                return idx + 1
    return -1
```

# Блок 8. Метрики для продакшен-системы

| Метрика | Цель | Красный уровень |
| :--- | :--- | :--- |
| **Average Generation Time** | Время одной генерации | > 60 секунд |
| **Queue Length** | Количество заданий в очереди | > 50 |
| **Worker Utilization** | Загрузка GPU | > 90% или < 20% (проблемы) |
| **Error Rate** | % неудачных генераций | > 5% |
| **Throughput** | Генераций в час | Мониторить тренд |

## 8.1. Скрипт для сбора метрик

```python
# metrics_collector.py
import requests
import json
import time
from datetime import datetime

def collect_comfyui_metrics():
    """Сбор метрик с ComfyUI"""
    try:
        # Статистика по очередям
        queue_response = requests.get("http://localhost:8188/queue")
        queue_data = queue_response.json()
        
        # Исторические данные
        history_response = requests.get("http://localhost:8188/history")
        history_data = history_response.json()
        
        metrics = {
            'timestamp': datetime.utcnow().isoformat(),
            'queue_pending': len(queue_data.get('queue_pending', [])),
            'queue_running': len(queue_data.get('queue_running', [])),
            'total_generations': len(history_data),
            'failed_generations': sum(1 for item in history_data.values() if 'error' in item)
        }
        
        # Сохранение в файл для последующей отправки
        with open('metrics.log', 'a') as f:
            f.write(json.dumps(metrics) + '\n')
        
        return metrics
    except Exception as e:
        print(f"Error collecting metrics: {e}")
        return None

if __name__ == "__main__":
    while True:
        collect_comfyui_metrics()
        time.sleep(60)
```

**Золотое правило ComfyUI в продакшене:**
> *"Всегда тестируйте новые workflow на небольшом батче, прежде чем запускать в массовое производство. Одна ошибка в ноде может стоить вам сотен GPU-часов."*

Начните с создания базового API-сервиса и одного рабочего, затем масштабируйте горизонтально. И помните: в ComfyUI важнее стабильность, чем скорость. Удачи в автоматизации генерации!
