# Блок 1. Архитектура видео-пайплайна

Современное видео в ComfyUI строится по принципу **конвейера**:

```
[Промпт] → [Генерация ключевых кадров] → [Интерполяция] → [Апскейлинг] → [Пост-обработка] → [Готовое видео]
```

**Компоненты:**
1. **Генератор** — SVD, AnimateDiff или LTX Video
2. **Интерполятор** — RIFE (кадры между кадрами)
3. **Апскейлер** — Real-ESRGAN / 4x-UltraSharp
4. **Временной фильтр** — для консистентности

# Блок 2. Сравнение моделей генерации (SVD vs AnimateDiff vs LTX)

| Характеристика | SVD (Stable Video Diffusion) | AnimateDiff | LTX Video |
| :--- | :--- | :--- | :--- |
| **Архитектура** | U-Net + Temporal Attention | Motion Module + SD1.5/SDXL | Transformer-based |
| **Длина видео** | 14-25 кадров (фиксировано) | 16-32 кадра (гибко) | До 256 кадров |
| **Качество** | Высокое (реалистичное) | Среднее (аниме/стилизованное) | Очень высокое (новый стандарт) |
| **Скорость** | Средняя (3-5 сек/кадр) | Быстрая (1-2 сек/кадр) | Медленная (но качественная) |
| **VRAM** | 12-16 GB | 8-12 GB | 20-24 GB |
| **Лучшее применение** | Реалистичные видео | Анимация, стилизация | Профессиональное видео |

## 3.3. Программная интерполяция (Python)

```python
# rife_interpolation.py
import cv2
import torch
from rife_models import RIFE
import numpy as np

class RIFEInterpolator:
    def __init__(self, model_path="rife-v4.6.pth", device="cuda"):
        self.model = RIFE(model_path)
        self.model = self.model.to(device)
        self.model.eval()
        self.device = device
    
    def interpolate_frames(self, frame1, frame2, multiplier=2):
        """
        Интерполяция между двумя кадрами
        multiplier = 2 → 1 промежуточный кадр
        multiplier = 4 → 3 промежуточных кадра
        """
        # Конвертация в тензоры
        frame1_t = torch.from_numpy(frame1).permute(2, 0, 1).float() / 255.0
        frame2_t = torch.from_numpy(frame2).permute(2, 0, 1).float() / 255.0
        
        frame1_t = frame1_t.unsqueeze(0).to(self.device)
        frame2_t = frame2_t.unsqueeze(0).to(self.device)
        
        # Генерация промежуточных кадров
        with torch.no_grad():
            if multiplier == 2:
                # Однократная интерполяция
                mid = self.model(frame1_t, frame2_t, timestep=0.5)
                frames = [frame1_t, mid, frame2_t]
            else:
                # Многократная интерполяция
                frames = [frame1_t]
                for i in range(1, multiplier):
                    t = i / multiplier
                    mid = self.model(frame1_t, frame2_t, timestep=t)
                    frames.append(mid)
                frames.append(frame2_t)
        
        # Конвертация обратно в numpy
        result = []
        for f in frames:
            f = f.squeeze(0).permute(1, 2, 0).cpu().numpy()
            f = (f * 255).astype(np.uint8)
            result.append(f)
        
        return result

# Использование
interpolator = RIFEInterpolator()
frame1 = cv2.imread("frame_001.png")
frame2 = cv2.imread("frame_002.png")
interpolated = interpolator.interpolate_frames(frame1, frame2, multiplier=4)
```

# batch_upscale.py
import os
import cv2
from realesrgan import RealESRGANer
from basicsr.archs.rrdbnet_arch import RRDBNet
import torch

def upscale_batch(input_dir, output_dir, scale=4):
    # Загрузка модели
    model = RRDBNet(num_in_ch=3, num_out_ch=3, num_feat=64, 
                    num_block=23, num_grow_ch=32, scale=scale)
    
    upsampler = RealESRGANer(
        scale=scale,
        model_path="weights/RealESRGAN_x4plus.pth",
        model=model,
        tile=512,
        tile_pad=16,
        pre_pad=0,
        half=False
    )
    
    # Обработка всех файлов
    os.makedirs(output_dir, exist_ok=True)
    files = sorted([f for f in os.listdir(input_dir) if f.endswith(('.png', '.jpg'))])
    
    for i, filename in enumerate(files):
        input_path = os.path.join(input_dir, filename)
        output_path = os.path.join(output_dir, f"upscaled_{i:04d}.png")
        
        img = cv2.imread(input_path, cv2.IMREAD_UNCHANGED)
        if img is not None:
            output, _ = upsampler.enhance(img, outscale=scale)
            cv2.imwrite(output_path, output)
            print(f"✅ Upscaled {filename}")
        else:
            print(f"❌ Failed to read {filename}")
    
    print(f"✅ Upscaled {len(files)} images")

# Использование
upscale_batch("frames/", "frames_upscaled/")

# Блок 5. Temporal Consistency (Временнáя консистентность)

Главная проблема AI-видео — мерцание и нестабильность между кадрами.

## 5.1. Temporal Filtering (Фильтр во времени)

```python
# temporal_filter.py
import numpy as np
import cv2
from scipy.ndimage import gaussian_filter1d

def apply_temporal_smoothing(frames, sigma=2.0):
    """
    Сглаживание кадров во времени
    frames: list of numpy arrays [H, W, C]
    sigma: степень сглаживания (2-5)
    """
    frames = np.array(frames)  # [T, H, W, C]
    smoothed = np.zeros_like(frames)
    
    # Применяем гауссово размытие по временной оси
    for c in range(frames.shape[3]):
        for h in range(frames.shape[1]):
            for w in range(frames.shape[2]):
                smoothed[:, h, w, c] = gaussian_filter1d(
                    frames[:, h, w, c], sigma=sigma, mode='reflect'
                )
    
    return [smoothed[t] for t in range(smoothed.shape[0])]

# Использование
frames = [cv2.imread(f"frame_{i:04d}.png") for i in range(50)]
smoothed = apply_temporal_smoothing(frames, sigma=3)
```

## 5.2. Optical Flow Stabilization

```python
# optical_flow_stabilization.py
import numpy as np
import cv2

def stabilize_video(frames, crop=30):
    """
    Стабилизация видео с использованием оптического потока
    """
    prev_gray = cv2.cvtColor(frames[0], cv2.COLOR_BGR2GRAY)
    transforms = []
    
    # Вычисление трансформаций
    for i in range(1, len(frames)):
        gray = cv2.cvtColor(frames[i], cv2.COLOR_BGR2GRAY)
        
        # Находим точки для трекинга
        prev_pts = cv2.goodFeaturesToTrack(prev_gray, maxCorners=200, qualityLevel=0.01, minDistance=30)
        curr_pts, status, _ = cv2.calcOpticalFlowPyrLK(prev_gray, gray, prev_pts, None)
        
        # Фильтруем валидные точки
        idx = np.where(status == 1)[0]
        prev_pts = prev_pts[idx]
        curr_pts = curr_pts[idx]
        
        # Вычисляем аффинное преобразование
        if len(prev_pts) > 10:
            m, _ = cv2.estimateAffinePartialTransform(prev_pts, curr_pts)
            transforms.append(m)
        else:
            transforms.append(np.eye(2, 3))
        
        prev_gray = gray
    
    # Применение трансформаций
    stabilized = []
    for i, frame in enumerate(frames):
        if i == 0:
            stabilized.append(frame[crop:-crop, crop:-crop])
        else:
            m = transforms[i-1]
            # Инвертируем трансформацию
            if m is not None:
                inv_m = cv2.invertAffineTransform(m)
                frame_stab = cv2.warpAffine(frame, inv_m, (frame.shape[1], frame.shape[0]))
                frame_stab = frame_stab[crop:-crop, crop:-crop]
                stabilized.append(frame_stab)
            else:
                stabilized.append(frame[crop:-crop, crop:-crop])
    
    return stabilized
```

# Блок 6. Batch Rendering (Масштабирование)

Для массовой генерации используйте **партионную обработку**.

## 6.1. Bash-скрипт для батч-рендеринга

```bash
#!/bin/bash
# batch_render.sh

INPUT_DIR="prompts/"
OUTPUT_DIR="videos/"
WORKFLOW="workflow_video.json"

mkdir -p $OUTPUT_DIR

for prompt_file in $INPUT_DIR/*.txt; do
    # Чтение промпта
    prompt=$(cat "$prompt_file")
    filename=$(basename "$prompt_file" .txt)
    
    echo " Rendering: $filename"
    
    # Замена промпта в JSON через jq
    jq --arg p "$prompt" \
       '.["5"].inputs.text = $p' \
       $WORKFLOW > temp_workflow.json
    
    # Отправка в ComfyUI API
    curl -X POST http://localhost:8188/prompt \
         -H "Content-Type: application/json" \
         -d @temp_workflow.json
    
    # Ожидание завершения
    sleep 30
    
    # Перемещение результата
    mv output/*.mp4 "$OUTPUT_DIR/${filename}.mp4"
    
    echo "✅ Done: $filename"
done

rm temp_workflow.json
```

## 6.2. Python-скрипт для распределенного рендеринга

```python
# distributed_render.py
import requests
import json
import time
from concurrent.futures import ThreadPoolExecutor
import os

class VideoRenderManager:
    def __init__(self, comfyui_urls=["http://localhost:8188"]):
        self.urls = comfyui_urls
        self.current_worker = 0
    
    def render(self, prompt_data, workflow_template):
        """Отправка задания на свободный воркер"""
        worker_url = self.urls[self.current_worker % len(self.urls)]
        self.current_worker += 1
        
        # Подстановка промпта
        workflow = json.loads(workflow_template)
        workflow["5"]["inputs"]["text"] = prompt_data["prompt"]
        
        # Отправка
        response = requests.post(f"{worker_url}/prompt", json={
            "prompt": workflow,
            "client_id": "batch_manager"
        })
        
        if response.status_code == 200:
            prompt_id = response.json()['prompt_id']
            return {
                "prompt_id": prompt_id,
                "worker": worker_url,
                "metadata": prompt_data
            }
        else:
            raise Exception(f"Error: {response.text}")
    
    def render_batch(self, prompt_list, workflow_template, max_workers=4):
        """Массовая генерация"""
        results = []
        
        with ThreadPoolExecutor(max_workers=max_workers) as executor:
            futures = [
                executor.submit(self.render, prompt, workflow_template)
                for prompt in prompt_list
            ]
            
            for future in futures:
                try:
                    result = future.result(timeout=300)
                    results.append(result)
                    print(f"✅ Задание отправлено: {result['metadata']['id']}")
                except Exception as e:
                    print(f"❌ Ошибка: {e}")
        
        return results
    
    def monitor_progress(self, results):
        """Мониторинг прогресса"""
        while True:
            completed = 0
            for result in results:
                status = self.get_status(result['prompt_id'], result['worker'])
                if status == 'completed':
                    completed += 1
            
            print(f"Прогресс: {completed}/{len(results)}")
            if completed == len(results):
                break
            time.sleep(10)
    
    def get_status(self, prompt_id, worker_url):
        """Проверка статуса задания"""
        response = requests.get(f"{worker_url}/history")
        if response.status_code == 200:
            history = response.json()
            if prompt_id in history:
                return 'completed'
        
        response = requests.get(f"{worker_url}/queue")
        if response.status_code == 200:
            queue = response.json()
            if any(prompt_id in item for item in queue.get('queue_running', [])):
                return 'running'
            if any(prompt_id in item for item in queue.get('queue_pending', [])):
                return 'pending'
        
        return 'not_found'

# Использование
manager = VideoRenderManager([
    "http://worker1:8188",
    "http://worker2:8188",
    "http://worker3:8188"
])

prompts = [
    {"id": 1, "prompt": "beautiful landscape with mountains"},
    {"id": 2, "prompt": "city at night with neon lights"},
    # ... еще 50 промптов
]

with open("workflow_template.json", "r") as f:
    workflow = f.read()

results = manager.render_batch(prompts, workflow, max_workers=3)
manager.monitor_progress(results)
```
