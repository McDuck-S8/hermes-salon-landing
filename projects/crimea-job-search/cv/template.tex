%% ─── Custom Commands ───

% Header with name and contact
\newcommand{\cvheader}[5]{
  \begin{center}
    {\Huge\bfseries\color{cvblue} #1}\\[0.3em]
    {\large\color{cvgray} #2}\\[0.5em]
    \small
    \color{cvgray}
    \hspace{0.5em}\faSymbol{@} #3%
    \hspace{1em}\faSymbol{phone} #4%
    \ifx&#5&\else\hspace{1em}\faSymbol{link} #5\fi
  \end{center}
  \vspace{0.5em}
}

% Skill inline list
\newcommand{\skilllist}[1]{
  \begin{center}
    \small\textbf{\color{cvblue}Навыки:}\enspace
    \color{cvgray}#1
  \end{center}
}

% Job entry
\newenvironment{jobentry}[4]{
  \subsection*{#1}
  {\small\color{cvgray}#2\enspace|\enspace#3\enspace|\enspace#4}\par
  \vspace{0.3em}
  \begin{itemize}[leftmargin=*,nosep,itemsep=0.1em]
}{
  \end{itemize}
}

% Education entry
\newcommand{\edentry}[4]{
  \subsection*{#1}
  {\small\color{cvgray}#2\enspace|\enspace#3\enspace–\enspace#4}\par
  \vspace{0.3em}
}

% Certification entry
\newcommand{\certentry}[2]{
  \item #1 \hfill \small\color{cvgray}#2
}
